ExplodeString(String:text[], String:split[], String:buffers[][], maxStrings, maxStringLength, bool:copyRemainder)
{
	new reloc_idx;
	new idx;
	new total;
	new var1;
	if (maxStrings < 1 || !split[0])
	{
		return 0;
	}
	while ((idx = SplitString(text[reloc_idx], split, buffers[total], maxStringLength)) != -1)
	{
		reloc_idx = idx + reloc_idx;
		total++;
		if (maxStrings == total)
		{
			if (copyRemainder)
			{
				strcopy(buffers[total + -1], maxStringLength, text[reloc_idx - idx]);
			}
			return total;
		}
	}
	total++;
	strcopy(buffers[total], maxStringLength, text[reloc_idx]);
	return total;
}

	l4d2_detonation_force_disable_gamemodes = CreateConVar("l4d2_detonation_force_disable_gamemodes", "empty", "Disable plugin in selected game modes", 320, false, 0.0, false, 0.0);
	AutoExecConfig(true, "l4d2_detonation_force", "sourcemod");

	GetGameFolderName(sGame, 256);
	if (StrEqual(sGame, "left4dead2", false))
	{
		isl4d2 = true;
		GameConf = LoadGameConfigFile("l4d2_detonationforce");
		if (!GameConf)
		{
			SetFailState("Couldn't find the offsets and signatures file. Please, check that it is installed correctly.");
		}
		StartPrepSDKCall(SDKCallType:2);
		PrepSDKCall_SetFromConf(GameConf, SDKFuncConfSource:1, "CTerrorPlayer_Fling");
		PrepSDKCall_AddParameter(SDKType:2, SDKPassMethod:3, 0, 0);
		PrepSDKCall_AddParameter(SDKType:4, SDKPassMethod:1, 0, 0);
		PrepSDKCall_AddParameter(SDKType:1, SDKPassMethod:0, 0, 0);
		PrepSDKCall_AddParameter(SDKType:5, SDKPassMethod:1, 0, 0);
		sdkCallPushPlayer = EndPrepSDKCall();
		if (!sdkCallPushPlayer)
		{
			SetFailState("Unable to find the 'CTerrorPlayer_Fling' signature, check the file version!");
		}
		CloseHandle(GameConf);
	}
	else
	{
		if (StrEqual(sGame, "left4dead", false))
		{
			isl4d2 = false;
		}
		SetFailState("L4D2 Explosion Force supports L4D and L4D 2 only");
	}
	HookConVarChange(l4d2_detonation_force_disable_gamemodes, VarGameModesChange);
	return void:0;
}

public LoadGameModes()
{
	new Handle:gamemode = FindConVar("mp_gamemode");
	static String:currentGameMode[64];
	GetConVarString(gamemode, currentGameMode, 64);
	static String:gamemodes[32][32];
	static String:buffer[512];
	GetConVarString(l4d2_detonation_force_disable_gamemodes, buffer, 512);
	ExplodeString(buffer, ",", gamemodes, 32, 32, false);
	if (StrEqual(buffer, "*", false))
	{
		PrintToServer("l4d2_DetonationForce: Disabled for all game modes");
		EnablePlugin = false;
		return 0;
	}
	EnablePlugin = true;
	new i;
	while (i < 32)
	{
		if (StrEqual(currentGameMode, gamemodes[i], false))
		{
			EnablePlugin = false;
			PrintToServer("l4d2_DetonationForce %s: Disabled", currentGameMode);
			if (EnablePlugin == true)
			{
				PrintToServer("l4d2_DetonationForce %s: Enabled", currentGameMode);
			}
			return 0;
		}
		i++;
	}
	if (EnablePlugin == true)
	{
		PrintToServer("l4d2_DetonationForce %s: Enabled", currentGameMode);
	}
	return 0;
}

public void:OnEntityDestroyed(entity)
{
	new var1;
	if (EnablePlugin && entity <= 0 && !IsValidEntity(entity) && !IsValidEdict(entity))
	{
		return void:0;
	}
	decl String:EntityName[128];
	if (!GetEdictClassname(entity, EntityName, 128))
	{
		return void:0;
	}
	new var2;
	if (StrEqual(EntityName, "prop_fuel_barrel", false) || StrEqual(EntityName, "pipe_bomb_projectile", false) || StrEqual(EntityName, "grenade_launcher_projectile", false))
	{
		decl Float:force;
		if (StrEqual(EntityName, "grenade_launcher_projectile", false))
		{
			force = float(GetConVarInt(l4d2_grenade_detonation_force));
		}
		else
		{
			if (StrEqual(EntityName, "prop_fuel_barrel", false))
			{
				force = float(GetConVarInt(l4d2_barrel_detonation_force));
			}
			if (GetEntityFlags(entity) == 1)
			{
				force = float(GetConVarInt(l4d2_pipe_detonation_force));
			}
			force = float(GetConVarInt(l4d2_bomb_detonation_force));
		}
		new client = 1;
		while (client <= MaxClients)
		{
			Fly(entity, client, force);
			client++;
		}
	}
	return void:0;
}

public Fly(explosion, target, Float:power)
{
	new var1;
	if (target <= 0 || !IsValidEntity(target) || !IsValidEdict(target))
	{
		return 0;
	}
	if (GetEntData(target, FindSendPropInfo("CTerrorPlayer", "m_isGhost", 0, 0, 0), 1) == 1)
	{
		if (!(GetConVarInt(l4d2_detonation_force_ghost_mode)))
		{
			return 0;
		}
	}
	decl Float:targetPos[3];
	decl Float:explosionPos[3];
	decl Float:traceVec[3];
	decl Float:resultingFling[3];
	GetEntPropVector(target, PropType:1, "m_vecOrigin", targetPos, 0);
	GetEntPropVector(explosion, PropType:1, "m_vecOrigin", explosionPos, 0);
	power -= GetVectorDistance(targetPos, explosionPos, false);
	if (power < 1.4E-45)
	{
		return 0;
	}
	MakeVectorFromPoints(explosionPos, targetPos, traceVec);
	GetVectorAngles(traceVec, resultingFling);
	resultingFling[0] = Cosine(DegToRad(resultingFling[1])) * power;
	resultingFling[1] = Sine(DegToRad(resultingFling[1])) * power;
	resultingFling[2] = power + power * 0.5;
	if (!isl4d2)
	{
		if (GetClientTeam(target) == 2)
		{
			new var2;
			if (GetConVarInt(l4d2_detonation_force_immunity) && GetConVarInt(l4d2_detonation_force_immunity) + 1 == 3)
			{
				DamageEffect(target, RoundToNearest(power * GetConVarFloat(l4d2_detonation_scale_damage)));
				TeleportEntity(target, NULL_VECTOR, NULL_VECTOR, resultingFling);
			}
		}
		else
		{
			if (GetClientTeam(target) == 3)
			{
				new var3;
				if (GetConVarInt(l4d2_detonation_force_immunity) && GetConVarInt(l4d2_detonation_force_immunity) + 1 == 2)
				{
					DamageEffect(target, RoundToNearest(power * GetConVarFloat(l4d2_detonation_scale_damage)));
					TeleportEntity(target, NULL_VECTOR, NULL_VECTOR, resultingFling);
				}
			}
		}
		return 0;
	}
	new var4;
	if (GetClientTeam(target) == 2 && IsVisibleTo(explosionPos, targetPos))
	{
		new var5;
		if (GetConVarInt(l4d2_detonation_force_immunity) && GetConVarInt(l4d2_detonation_force_immunity) + 1 == 3)
		{
			DamageEffect(target, RoundToNearest(power * GetConVarFloat(l4d2_detonation_scale_damage)));
			SDKCall(sdkCallPushPlayer, target, resultingFling, 76, target, 1073741824);
		}
	}
	else
	{
		if (GetClientTeam(target) == 3)
		{
			new var6;
			if (GetConVarInt(l4d2_detonation_force_immunity) && GetConVarInt(l4d2_detonation_force_immunity) + 1 == 2)
			{
				DamageEffect(target, RoundToNearest(power * GetConVarFloat(l4d2_detonation_scale_damage)));
				SDKCall(sdkCallPushPlayer, target, resultingFling, 2, target, 1073741824);
			}
		}
	}
	return 0;
}

bool:IsVisibleTo(Float:position[3], Float:targetposition[3])
{
	decl Float:vAngles[3];
	decl Float:vLookAt[3];
	MakeVectorFromPoints(position, targetposition, vLookAt);
	GetVectorAngles(vLookAt, vAngles);
	new Handle:trace = TR_TraceRayFilterEx(position, vAngles, 16395, RayType:1, _TraceFilter, any:0);
	new bool:isVisible;
	if (TR_DidHit(trace))
	{
		decl Float:vStart[3];
		TR_GetEndPosition(vStart, trace);
		if (GetVectorDistance(position, vStart, false) + 25.0 >= GetVectorDistance(position, targetposition, false))
		{
			isVisible = true;
		}
	}
	else
	{
		LogError("Tracer Bug: Player-Zombie Trace did not hit anything, WTF");
		isVisible = true;
	}
	CloseHandle(trace);
	return isVisible;
}
